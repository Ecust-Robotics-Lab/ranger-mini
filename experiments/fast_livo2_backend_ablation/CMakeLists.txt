cmake_minimum_required(VERSION 3.16)
project(p0_backend_ablation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ranger_fast_livo_backend REQUIRED)
add_executable(graph_ablation graph_ablation.cpp)
target_link_libraries(graph_ablation ranger_fast_livo_backend::ranger_backend_core)
add_executable(backend_ablation backend_ablation.cpp)
target_link_libraries(backend_ablation ranger_fast_livo_backend::ranger_backend_core)
option(RANGER_PATH_FILTER_ENABLED "Compile against the isolated path-filter backend ABI" OFF)
if(RANGER_PATH_FILTER_ENABLED)
  target_compile_definitions(backend_ablation PRIVATE RANGER_PATH_FILTER_ENABLED)
endif()
