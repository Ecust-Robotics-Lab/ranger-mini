cmake_minimum_required(VERSION 3.16)
project(ranger_slam_reader)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
find_package(glog REQUIRED)
add_executable(check_lightning_bag check_lightning_bag.cc)
target_include_directories(check_lightning_bag PRIVATE
  "${ROOT}/ws_slam/src/lightning/src" "${ROOT}/ws_slam/src/lightning/thirdparty" ${PCL_INCLUDE_DIRS})
ament_target_dependencies(check_lightning_bag rclcpp tf2 tf2_ros rosbag2_cpp nav_msgs sensor_msgs livox_ros_driver2 pcl_conversions)
target_link_libraries(check_lightning_bag "${ROOT}/ws_slam/build/lightning/src/liblightning.libs.so" glog::glog ${PCL_LIBRARIES})
