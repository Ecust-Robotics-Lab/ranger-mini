cmake_minimum_required(VERSION 3.16)
project(ranger_fast_livo_localizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(ranger_fast_livo_backend REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/LocalizerEstimate.msg" "srv/LoadMap.srv" "srv/Initialize.srv" "srv/Relocalize.srv"
  DEPENDENCIES std_msgs geometry_msgs)
option(RANGER_ENABLE_KISS "Explicitly enable optional C2 KISS-Matcher (includes GPLv3+ PMC)" OFF)
add_library(ranger_localizer_core src/archive_reader.cpp src/localizer.cpp src/place_index.cpp src/relocalizer.cpp src/global_coarse.cpp)
target_include_directories(ranger_localizer_core PUBLIC include)
target_link_libraries(ranger_localizer_core ranger_fast_livo_backend::ranger_backend_core)
ament_target_dependencies(ranger_localizer_core ranger_fast_livo_backend)
if(RANGER_ENABLE_KISS)
  find_package(TBB REQUIRED)
  find_package(robin REQUIRED)
  find_package(kiss_matcher REQUIRED)
  target_compile_definitions(ranger_localizer_core PUBLIC RANGER_ENABLE_KISS)
  target_link_libraries(ranger_localizer_core kiss_matcher::kiss_matcher_core)
endif()
target_compile_options(ranger_localizer_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(localizer_node src/node.cpp)
ament_target_dependencies(localizer_node rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs diagnostic_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(localizer_node ranger_localizer_core "${cpp_typesupport_target}")
add_executable(archive_tool src/archive_tool.cpp)
target_link_libraries(archive_tool ranger_localizer_core)
add_executable(place_index_tool src/place_index_tool.cpp)
target_link_libraries(place_index_tool ranger_localizer_core)
add_executable(tag_map_tool src/tag_map_tool.cpp)
target_link_libraries(tag_map_tool ranger_localizer_core)
install(TARGETS localizer_node archive_tool place_index_tool tag_map_tool DESTINATION lib/${PROJECT_NAME})
install(TARGETS ranger_localizer_core DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY third_party DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(recovery_replay test/recovery_replay.cpp)
  target_link_libraries(recovery_replay ranger_localizer_core)
  add_executable(selection_experiment test/selection_experiment.cpp)
  target_link_libraries(selection_experiment ranger_localizer_core)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(localizer_tests test/localizer_tests.cpp TIMEOUT 180)
  target_link_libraries(localizer_tests ranger_localizer_core)
  ament_add_gtest(tag_integration_tests test/tag_integration_tests.cpp TIMEOUT 180)
  target_link_libraries(tag_integration_tests ranger_localizer_core)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
