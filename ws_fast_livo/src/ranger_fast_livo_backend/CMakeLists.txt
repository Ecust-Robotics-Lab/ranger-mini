cmake_minimum_required(VERSION 3.16)
project(ranger_fast_livo_backend LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(GTSAM 4.2 REQUIRED CONFIG)
find_package(small_gicp REQUIRED CONFIG)
find_package(PCL 1.12 REQUIRED COMPONENTS common io filters kdtree registration features)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/BackendEstimate.msg" "srv/ExportSnapshot.srv"
  DEPENDENCIES std_msgs geometry_msgs)
add_library(ranger_backend_core src/input.cpp src/registration.cpp src/graph.cpp src/backend.cpp src/archive.cpp)
target_include_directories(ranger_backend_core PUBLIC include ${PCL_INCLUDE_DIRS})
target_link_libraries(ranger_backend_core Eigen3::Eigen gtsam small_gicp::small_gicp ${PCL_LIBRARIES} OpenSSL::Crypto nlohmann_json::nlohmann_json)
ament_target_dependencies(ranger_backend_core rclcpp nav_msgs sensor_msgs pcl_conversions)
target_compile_options(ranger_backend_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(backend_node src/node.cpp)
ament_target_dependencies(backend_node rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs diagnostic_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(backend_node ranger_backend_core "${cpp_typesupport_target}")
install(TARGETS backend_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS ranger_backend_core DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY third_party/ DESTINATION share/${PROJECT_NAME}/third_party)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(backend_tests test/backend_tests.cpp TIMEOUT 180)
  target_link_libraries(backend_tests ranger_backend_core)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
